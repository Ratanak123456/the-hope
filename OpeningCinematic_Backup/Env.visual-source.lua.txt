--!nonstrict
-- "The Guardian Beneath the Ice": the North Pole environment. Client-local
-- only, built far above the real playable map - the same "tucked away in a
-- box, never seen clipping through real city geometry, never replicates"
-- pattern MenuScene.lua already uses for the menu hangar (see its header
-- comment). Built once when the cinematic starts, destroyed the moment it
-- ends (Skip or natural completion) - see Opening.lua.
--
-- Layout runs along -Z as the story goes deeper: arrival convoy (+Z) ->
-- research base -> iceberg excavation face (Z=0) -> ice tunnel -> the
-- ancient door -> the Aegis chamber -> a shaft down to the alien prison
-- (lowest Y). Everything is anchored, non-colliding, non-querying decorative
-- geometry - nothing here needs physics since the whole cinematic is
-- camera-and-CFrame driven (rule 18: "use scripted character paths").

local Kit = require(script.Parent.Kit)
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local Palette = require(ReplicatedStorage:WaitForChild("Shared").Palette)

local Env = {}

local ORIGIN = Vector3.new(0, 9500, 0)

Env.Zones = {
	ConvoyFar = ORIGIN + Vector3.new(10, 26, 190),
	ConvoyNear = ORIGIN + Vector3.new(0, 4, 90),
	BaseCenter = ORIGIN + Vector3.new(4, 4, 46),
	IcebergFace = ORIGIN + Vector3.new(0, 6, -2),
	TunnelEntrance = ORIGIN + Vector3.new(0, 2, -8),
	Door = ORIGIN + Vector3.new(0, -13, -44),
	HallCenter = ORIGIN + Vector3.new(0, -20, -66),
	ChamberCenter = ORIGIN + Vector3.new(0, -32, -98),
	ChamberFloor = ORIGIN + Vector3.new(0, -40, -98),
	ShaftTop = ORIGIN + Vector3.new(0, -42, -118),
	PrisonCenter = ORIGIN + Vector3.new(0, -92, -150),
}

local function marker(markers: Folder, name: string, cframe: CFrame)
	local part = Kit.part({ name = name, size = Vector3.new(0.2, 0.2, 0.2), color = Color3.new(1, 0, 1), transparency = 1, cframe = cframe })
	part.Parent = markers
end

local function tent(parent: Instance, position: Vector3, size: Vector3, color: Color3, rng: Random)
	local base = Kit.wedge({ name = "TentPanel", size = Vector3.new(size.X, size.Y, size.Z * 0.5), color = color, material = Enum.Material.Fabric, cframe = CFrame.new(position) * CFrame.Angles(0, rng:NextNumber() * math.pi * 2, 0), shadow = true })
	base.Parent = parent
	local mirror = Kit.wedge({ name = "TentPanel2", size = Vector3.new(size.X, size.Y, size.Z * 0.5), color = color, material = Enum.Material.Fabric, cframe = base.CFrame * CFrame.new(0, 0, size.Z * 0.25) * CFrame.Angles(0, math.pi, 0) * CFrame.new(0, 0, size.Z * 0.25), shadow = true })
	mirror.Parent = parent
	local doorFlap = Kit.part({ name = "TentDoor", size = Vector3.new(size.X * 0.32, size.Y * 0.7, 0.1), color = Palette.Ancient.StoneDark, material = Enum.Material.Fabric, cframe = base.CFrame * CFrame.new(0, -size.Y * 0.15, size.Z * 0.26) })
	doorFlap.Parent = parent
	return base
end

local function crate(parent: Instance, cframe: CFrame, size: Vector3, color: Color3?)
	local box = Kit.part({ name = "Crate", size = size, color = color or Color3.fromRGB(90, 78, 56), material = Enum.Material.WoodPlanks, cframe = cframe, shadow = true })
	box.Parent = parent
	return box
end

local function buildGround(parent: Instance, rng: Random)
	local snowTop = Color3.fromRGB(214, 222, 228)
	local snowShadow = Color3.fromRGB(176, 186, 196)

	-- Wide, tiled snowfield rather than one giant flat plate - lets height
	-- vary block to block, which reads as rolling snowbanks from a distance.
	for gx = -3, 3 do
		for gz = -1, 6 do
			local cx = gx * 60
			local cz = gz * 60 - 40
			local height = 3 + rng:NextNumber() * 2.5
			local part = Kit.part({
				name = "SnowTile",
				size = Vector3.new(60, height, 60),
				color = if rng:NextNumber() < 0.3 then snowShadow else snowTop,
				material = Enum.Material.Snow,
				cframe = CFrame.new(ORIGIN + Vector3.new(cx, -height / 2, cz)),
			})
			part.Parent = parent
		end
	end

	-- Snowbank mounds and wind-carved ridges near the base and excavation.
	for i = 1, 14 do
		local pos = ORIGIN + Vector3.new(rng:NextNumber(-130, 130), rng:NextNumber(1, 4), rng:NextNumber(-30, 160))
		local mound = Kit.ball({ name = "Snowbank", size = Vector3.new(rng:NextNumber(6, 16), rng:NextNumber(3, 6), rng:NextNumber(6, 16)), color = snowTop, material = Enum.Material.Snow, cframe = CFrame.new(pos) })
		mound.Parent = parent
	end

	-- Cracked ice surfaces: thin dark inset lines near the excavation.
	for i = 1, 8 do
		local pos = ORIGIN + Vector3.new(rng:NextNumber(-24, 24), 4.4, rng:NextNumber(-20, 30))
		local crack = Kit.part({ name = "IceCrack", size = Vector3.new(rng:NextNumber(4, 12), 0.08, rng:NextNumber(0.3, 0.6)), color = Color3.fromRGB(70, 90, 100), material = Enum.Material.Ice, transparency = 0.2, cframe = CFrame.new(pos) * CFrame.Angles(0, rng:NextNumber() * math.pi, 0) })
		crack.Parent = parent
	end

	-- Wind-driven snow close to the ground - one restrained emitter (matches
	-- MenuScene's "one particle system, nothing more" budget) on an
	-- invisible part low over the base, wide enough to read across the
	-- whole exterior without needing one emitter per shot.
	local windPart = Kit.part({ name = "SnowWindEmitter", size = Vector3.new(90, 1, 90), color = Color3.new(1, 1, 1), transparency = 1, cframe = CFrame.new(Env.Zones.BaseCenter + Vector3.new(0, 2, 0)) })
	windPart.Parent = parent
	local snow = Instance.new("ParticleEmitter")
	snow.Color = ColorSequence.new(Color3.fromRGB(235, 240, 245))
	snow.Rate = 40
	snow.Lifetime = NumberRange.new(2.5, 4)
	snow.Speed = NumberRange.new(14, 22)
	snow.SpreadAngle = Vector2.new(20, 8)
	snow.Size = NumberSequence.new(0.12)
	snow.Transparency = NumberSequence.new({ NumberSequenceKeypoint.new(0, 0.3), NumberSequenceKeypoint.new(1, 1) })
	snow.Acceleration = Vector3.new(-6, -1, 0)
	snow.Parent = windPart

	-- Dark broken ocean water, visible far to one side beyond the ice shelf.
	local water = Kit.part({ name = "BrokenSea", size = Vector3.new(140, 1, 90), color = Color3.fromRGB(18, 26, 34), material = Enum.Material.Ice, transparency = 0.15, cframe = CFrame.new(ORIGIN + Vector3.new(-170, 3.4, 60)) })
	water.Parent = parent
	for i = 1, 5 do
		local floe = Kit.part({ name = "IceFloe", size = Vector3.new(rng:NextNumber(8, 18), 1, rng:NextNumber(8, 16)), color = Color3.fromRGB(206, 214, 220), material = Enum.Material.Ice, cframe = CFrame.new(water.Position + Vector3.new(rng:NextNumber(-60, 60), 0.6, rng:NextNumber(-30, 30))) * CFrame.Angles(0, rng:NextNumber() * math.pi, 0) })
		floe.Parent = parent
	end
end

local function buildMountains(parent: Instance, rng: Random)
	local colors = { Color3.fromRGB(150, 158, 168), Color3.fromRGB(170, 176, 182), Color3.fromRGB(190, 194, 198) }
	for i = 1, 10 do
		local angle = (i / 10) * math.pi * 2
		local dist = 260 + rng:NextNumber(0, 40)
		local pos = ORIGIN + Vector3.new(math.cos(angle) * dist, 30 + rng:NextNumber(0, 24), math.sin(angle) * dist + 30)
		local ridge = Kit.part({ name = "Mountain", size = Vector3.new(rng:NextNumber(50, 90), rng:NextNumber(50, 110), 20), color = colors[(i % 3) + 1], material = Enum.Material.Slate, transparency = 0.25 + rng:NextNumber(0, 0.25), cframe = CFrame.new(pos, ORIGIN + Vector3.new(0, 40, 30)) })
		ridge.CastShadow = false
		ridge.Parent = parent
	end
end

local function buildIceberg(parent: Instance, rng: Random): BasePart
	local iceColorOuter = Color3.fromRGB(224, 234, 238)
	local iceColorDeep = Color3.fromRGB(120, 168, 186)

	local face: BasePart? = nil
	for i = 1, 16 do
		local offset = Vector3.new(rng:NextNumber(-46, 46), rng:NextNumber(-4, 44), rng:NextNumber(-70, 10))
		local size = Vector3.new(rng:NextNumber(14, 30), rng:NextNumber(20, 60), rng:NextNumber(14, 30))
		local chunk = Kit.part({
			name = "IcebergChunk",
			size = size,
			color = if rng:NextNumber() < 0.65 then iceColorOuter else iceColorDeep,
			material = Enum.Material.Ice,
			transparency = if rng:NextNumber() < 0.5 then 0 else 0.25,
			cframe = CFrame.new(Env.Zones.IcebergFace + offset) * CFrame.Angles(rng:NextNumber(-0.05, 0.05), rng:NextNumber() * math.pi, rng:NextNumber(-0.05, 0.05)),
			shadow = true,
		})
		chunk.Parent = parent
		if offset.Z > -6 and offset.Z < 4 and not face then
			face = chunk
		end
	end

	-- Faint straight lines inside the ice hinting at buried architecture -
	-- darker, more opaque inset slabs, never a full reveal.
	for i = 1, 5 do
		local pos = Env.Zones.IcebergFace + Vector3.new(rng:NextNumber(-20, 20), rng:NextNumber(6, 30), rng:NextNumber(-40, -10))
		local hint = Kit.part({ name = "BuriedStructureHint", size = Vector3.new(1, rng:NextNumber(10, 22), 1), color = Color3.fromRGB(70, 90, 96), material = Enum.Material.Ice, transparency = 0.55, cframe = CFrame.new(pos) })
		hint.Parent = parent
	end

	-- The excavation opening itself: a dark, human-cut rectangular mouth in
	-- the ice face, framed by temporary metal shoring.
	local mouth = Kit.part({ name = "ExcavationMouth", size = Vector3.new(9, 10, 4), color = Color3.fromRGB(10, 14, 16), material = Enum.Material.Slate, transparency = 0.15, cframe = CFrame.new(Env.Zones.TunnelEntrance) })
	mouth.Parent = parent
	for _, side in { -1, 1 } do
		local beam = Kit.part({ name = "ShoringBeam", size = Vector3.new(1, 11, 1.2), color = Palette.City.Equipment, material = Enum.Material.Metal, cframe = CFrame.new(Env.Zones.TunnelEntrance + Vector3.new(side * 4.6, 0.5, 2)) })
		beam.Parent = parent
	end
	local lintel = Kit.part({ name = "ShoringLintel", size = Vector3.new(10.4, 1, 1.2), color = Palette.City.Equipment, material = Enum.Material.Metal, cframe = CFrame.new(Env.Zones.TunnelEntrance + Vector3.new(0, 5.5, 2)) })
	lintel.Parent = parent

	return face or mouth
end

local function buildVehicle(parent: Instance, cframe: CFrame, kind: "Truck" | "Tracked"): { model: Model, headlights: { BasePart } }
	local model = Instance.new("Model")
	model.Name = kind .. "Vehicle"
	local bodySize = if kind == "Truck" then Vector3.new(6, 3.4, 11) else Vector3.new(7, 3.8, 13)
	local body = Kit.part({ name = "Body", size = bodySize, color = Color3.fromRGB(70, 76, 62), material = Enum.Material.Metal, cframe = cframe, shadow = true })
	body.Parent = model
	local cab = Kit.part({ name = "Cab", size = Vector3.new(bodySize.X * 0.9, 1.6, 3.4), color = Color3.fromRGB(50, 54, 46), material = Enum.Material.Metal, cframe = cframe * CFrame.new(0, bodySize.Y * 0.5 + 0.8, bodySize.Z * 0.28) })
	cab.Parent = model
	local windshield = Kit.part({ name = "Windshield", size = Vector3.new(bodySize.X * 0.82, 1.1, 0.1), color = Palette.City.Glass, material = Enum.Material.Glass, transparency = 0.3, cframe = cab.CFrame * CFrame.new(0, 0.1, 1.7) })
	windshield.Parent = model

	local headlights: { BasePart } = {}
	for _, side in { -1, 1 } do
		local lamp = Kit.part({ name = "Headlight", size = Vector3.new(0.6, 0.6, 0.2), color = Color3.fromRGB(255, 244, 210), material = Enum.Material.Neon, cframe = cframe * CFrame.new(side * bodySize.X * 0.35, -bodySize.Y * 0.1, bodySize.Z * 0.5) })
		lamp.Parent = model
		local spot = Instance.new("SpotLight")
		spot.Color = Color3.fromRGB(255, 240, 210)
		spot.Range = 45
		spot.Brightness = 3
		spot.Angle = 45
		spot.Face = Enum.NormalId.Front
		spot.Parent = lamp
		table.insert(headlights, lamp)
	end

	if kind == "Truck" then
		for _, side in { -1, 1 } do
			for _, dz in { -1, 1 } do
				local wheel = Kit.cylinder({ name = "Wheel", size = Vector3.new(1.6, 1.4, 1.4), color = Color3.fromRGB(18, 18, 20), material = Enum.Material.Metal, cframe = cframe * CFrame.new(side * bodySize.X * 0.5, -bodySize.Y * 0.4, dz * bodySize.Z * 0.32) }, "z")
				wheel.Parent = model
			end
		end
	else
		for _, side in { -1, 1 } do
			local track = Kit.part({ name = "Track", size = Vector3.new(1.3, 1.6, bodySize.Z * 0.95), color = Color3.fromRGB(22, 22, 24), material = Enum.Material.Metal, cframe = cframe * CFrame.new(side * bodySize.X * 0.52, -bodySize.Y * 0.35, 0) })
			track.Parent = model
		end
	end

	local antenna = Kit.part({ name = "Antenna", size = Vector3.new(0.1, 3, 0.1), color = Color3.fromRGB(40, 40, 44), material = Enum.Material.Metal, cframe = cab.CFrame * CFrame.new(bodySize.X * 0.3, 2.2, -1) })
	antenna.Parent = model

	model.Parent = parent
	return { model = model, headlights = headlights }
end

local function buildResearchBase(parent: Instance, rng: Random): (Model, BasePart, { PointLight }, { SpotLight }, { { model: Model, headlights: { BasePart } } })
	local base = Env.Zones.BaseCenter

	tent(parent, base + Vector3.new(-18, 4.5, 6), Vector3.new(14, 9, 18), Color3.fromRGB(70, 78, 60), rng)
	tent(parent, base + Vector3.new(-20, 3.6, 26), Vector3.new(10, 7, 12), Color3.fromRGB(200, 60, 60), rng) -- medical, red cross reads by colour alone

	-- Generator station.
	local generator = Kit.part({ name = "Generator", size = Vector3.new(6, 4, 5), color = Palette.City.Equipment, material = Enum.Material.Metal, cframe = CFrame.new(base + Vector3.new(6, 6, 4)) })
	generator.Parent = parent
	Kit.light(generator, Color3.fromRGB(255, 210, 140), 14, 1.6)

	-- Communication tower - destroyed later, so kept as its own Model.
	local commTower = Instance.new("Model")
	commTower.Name = "CommTower"
	local towerBase = Kit.part({ name = "TowerBase", size = Vector3.new(3, 1, 3), color = Palette.City.Equipment, material = Enum.Material.Metal, cframe = CFrame.new(base + Vector3.new(20, 1, -6)) })
	towerBase.Parent = commTower
	local towerMast = Kit.part({ name = "TowerMast", size = Vector3.new(1, 26, 1), color = Color3.fromRGB(150, 152, 156), material = Enum.Material.Metal, cframe = towerBase.CFrame * CFrame.new(0, 13.5, 0), shadow = true })
	towerMast.Parent = commTower
	for i = 1, 3 do
		local dish = Kit.part({ name = `Dish{i}`, size = Vector3.new(2.4, 2.4, 0.3), color = Color3.fromRGB(200, 202, 206), material = Enum.Material.Metal, cframe = towerMast.CFrame * CFrame.new(0, 9 - i * 3, 0) * CFrame.Angles(0, i * 1.1, 0) })
		dish.Parent = commTower
	end
	local towerLight = Kit.ball({ name = "TowerLight", size = Vector3.new(0.6, 0.6, 0.6), color = Color3.fromRGB(255, 60, 40), material = Enum.Material.Neon, cframe = towerMast.CFrame * CFrame.new(0, 13.5, 0) })
	towerLight.Parent = commTower
	commTower.Parent = parent

	-- Radar/satellite dish, kept separate so Sequences can spin it.
	local radarDish = Kit.part({ name = "RadarDish", size = Vector3.new(5, 0.4, 5), color = Color3.fromRGB(210, 212, 214), material = Enum.Material.Metal, cframe = CFrame.new(base + Vector3.new(28, 5, 8)) * CFrame.Angles(math.rad(35), 0, 0) })
	radarDish.Parent = parent
	local radarMount = Kit.part({ name = "RadarMount", size = Vector3.new(1, 5, 1), color = Palette.City.Equipment, material = Enum.Material.Metal, cframe = CFrame.new(base + Vector3.new(28, 2.5, 8)) })
	radarMount.Parent = parent

	-- Excavation machinery: drill rig facing the iceberg.
	local rigBase = Kit.part({ name = "DrillRigBase", size = Vector3.new(8, 2, 8), color = Palette.City.Equipment, material = Enum.Material.DiamondPlate, cframe = CFrame.new(base + Vector3.new(-2, 1, -18)) })
	rigBase.Parent = parent
	local rigTower = Kit.part({ name = "DrillRigTower", size = Vector3.new(2, 20, 2), color = Color3.fromRGB(180, 100, 40), material = Enum.Material.Metal, cframe = rigBase.CFrame * CFrame.new(0, 11, 0) * CFrame.Angles(math.rad(-8), 0, 0), shadow = true })
	rigTower.Parent = parent
	local drillBit = Kit.cylinder({ name = "DrillBit", size = Vector3.new(1.2, 6, 1.2), color = Color3.fromRGB(90, 92, 96), material = Enum.Material.CorrodedMetal, cframe = rigTower.CFrame * CFrame.new(0, -9, 0) }, "y")
	drillBit.Parent = parent
	local iceDust = Instance.new("ParticleEmitter")
	iceDust.Color = ColorSequence.new(Color3.fromRGB(220, 228, 232))
	iceDust.Rate = 18
	iceDust.Lifetime = NumberRange.new(0.6, 1.2)
	iceDust.Speed = NumberRange.new(2, 5)
	iceDust.SpreadAngle = Vector2.new(35, 35)
	iceDust.Size = NumberSequence.new(0.18)
	iceDust.Transparency = NumberSequence.new({ NumberSequenceKeypoint.new(0, 0.2), NumberSequenceKeypoint.new(1, 1) })
	iceDust.Parent = drillBit

	-- Cargo containers, fuel storage, equipment tables + non-readable
	-- displays, portable lights, warning barriers, cabling.
	local portableLights: { SpotLight } = {}
	for i = 1, 5 do
		local pos = base + Vector3.new(rng:NextNumber(-34, 34), 0, rng:NextNumber(-24, 40))
		crate(parent, CFrame.new(pos + Vector3.new(0, 2, 0)), Vector3.new(rng:NextNumber(4, 7), 4, rng:NextNumber(4, 7)), if rng:NextNumber() < 0.5 then Color3.fromRGB(90, 78, 56) else Color3.fromRGB(60, 66, 58))
	end
	for i = 1, 2 do
		local tank = Kit.cylinder({ name = "FuelTank", size = Vector3.new(3.2, 6, 3.2), color = Color3.fromRGB(150, 60, 50), material = Enum.Material.Metal, cframe = CFrame.new(base + Vector3.new(-30 + i * 4, 3, 30)) }, "y")
		tank.Parent = parent
	end
	for i = 1, 6 do
		local angle = rng:NextNumber() * math.pi * 2
		local dist = rng:NextNumber(20, 44)
		local pos = base + Vector3.new(math.cos(angle) * dist, 5, math.sin(angle) * dist)
		local pole = Kit.part({ name = "LightPole", size = Vector3.new(0.4, 10, 0.4), color = Color3.fromRGB(60, 62, 66), material = Enum.Material.Metal, cframe = CFrame.new(pos) })
		pole.Parent = parent
		local fixture = Kit.part({ name = "LightFixture", size = Vector3.new(1.2, 0.5, 1.2), color = Color3.fromRGB(40, 40, 44), material = Enum.Material.Metal, cframe = pole.CFrame * CFrame.new(0, 5, 0) })
		fixture.Parent = parent
		local spot = Instance.new("SpotLight")
		spot.Color = Color3.fromRGB(255, 236, 200)
		spot.Range = 30
		spot.Brightness = 2.2
		spot.Angle = 70
		spot.Face = Enum.NormalId.Bottom
		spot.Parent = fixture
		table.insert(portableLights, spot)
	end

	-- Warning barriers + tape line near the excavation mouth.
	for i = -2, 2 do
		local barrier = Kit.part({ name = "Barrier", size = Vector3.new(3, 1.1, 0.3), color = Color3.fromRGB(220, 170, 40), material = Enum.Material.SmoothPlastic, cframe = CFrame.new(Env.Zones.TunnelEntrance + Vector3.new(i * 3.4, -3.8, 8)) })
		barrier.Parent = parent
	end

	-- Equipment table with a non-readable scientific display (a lit panel
	-- with abstract inset shapes, never real text).
	local table_ = Kit.part({ name = "EquipmentTable", size = Vector3.new(4, 1.4, 2), color = Palette.City.Equipment, material = Enum.Material.Metal, cframe = CFrame.new(base + Vector3.new(-6, 0.7, 16)) })
	table_.Parent = parent
	local display = Kit.part({ name = "Display", size = Vector3.new(2.4, 1.5, 0.15), color = Palette.Energy.CyanDim, material = Enum.Material.Neon, transparency = 0.1, cframe = table_.CFrame * CFrame.new(0, 1.3, -0.5) * CFrame.Angles(math.rad(-60), 0, 0) })
	display.Parent = parent

	-- Cables, partially covered by snow, running between generator and tents.
	for i = 1, 4 do
		local from = generator.Position
		local to = base + Vector3.new(rng:NextNumber(-20, 0), 0, rng:NextNumber(0, 26))
		local mid = from:Lerp(to, 0.5)
		local cable = Kit.part({ name = "Cable", size = Vector3.new(0.25, 0.15, (to - from).Magnitude), color = Color3.fromRGB(20, 20, 22), material = Enum.Material.SmoothPlastic, cframe = CFrame.lookAt(mid, to) })
		cable.Parent = parent
	end

	-- Red warning lights, visible through the storm; collected for strobing.
	local redLights: { PointLight } = {}
	for i = 1, 4 do
		local pos = base + Vector3.new(rng:NextNumber(-30, 30), 6, rng:NextNumber(-10, 34))
		local housing = Kit.part({ name = "WarningLight", size = Vector3.new(0.6, 0.6, 0.6), color = Color3.fromRGB(200, 40, 30), material = Enum.Material.Neon, cframe = CFrame.new(pos) })
		housing.Parent = parent
		local light = Kit.light(housing, Color3.fromRGB(255, 50, 40), 20, 0)
		if light then
			table.insert(redLights, light)
		end
	end

	-- Vehicles: a small convoy plus two parked tracked transports at the base.
	local vehicles = {}
	table.insert(vehicles, buildVehicle(parent, CFrame.new(Env.Zones.ConvoyFar) * CFrame.Angles(0, math.pi, 0), "Truck"))
	table.insert(vehicles, buildVehicle(parent, CFrame.new(Env.Zones.ConvoyFar + Vector3.new(8, 0, 14)) * CFrame.Angles(0, math.pi, 0), "Tracked"))
	table.insert(vehicles, buildVehicle(parent, CFrame.new(Env.Zones.ConvoyFar + Vector3.new(-9, 0, 26)) * CFrame.Angles(0, math.pi, 0), "Truck"))
	table.insert(vehicles, buildVehicle(parent, CFrame.new(base + Vector3.new(16, 0, -8)) * CFrame.Angles(0, math.rad(160), 0), "Tracked"))

	return commTower, radarDish, redLights, portableLights, vehicles
end

--------------------------------------------------------------------------------
-- Command tent interior (Sequence 2) - a distinct dressed interior, built
-- inside the command tent's footprint.
--------------------------------------------------------------------------------

local function buildCommandInterior(parent: Instance): BasePart
	local center = Env.Zones.BaseCenter + Vector3.new(-18, 0, 6)

local function buildTunnel(parent: Instance, rng: Random)
	local from = Env.Zones.TunnelEntrance
	local to = Env.Zones.Door
	local steps = 8
	for i = 0, steps - 1 do
		local t0 = i / steps
		local t1 = (i + 1) / steps
		local p0 = from:Lerp(to, t0)
		local p1 = from:Lerp(to, t1)
		local mid = p0:Lerp(p1, 0.5)
		local wallColor = Color3.fromRGB(150, 176, 186):Lerp(Palette.Ancient.StonePale, t0) -- gradually shifts from ice to ancient stone
		-- The camera path runs vertically centred inside this shell, so
		-- travelling it reads as walking through a tunnel interior (Roblox
		-- parts render their inside faces too - no separate wall+wall+ceiling
		-- assembly needed for a straight corridor segment).
		local tunnelSeg = Kit.part({ name = `TunnelSeg{i}`, size = Vector3.new(9, 8.5, (p1 - p0).Magnitude + 0.4), color = wallColor, material = if t0 < 0.4 then Enum.Material.Ice else Enum.Material.Slate, cframe = CFrame.lookAt(mid, p1), shadow = true })
		tunnelSeg.Parent = parent

		local floorSeg = Kit.part({ name = "TunnelFloor", size = Vector3.new(8, 0.4, (p1 - p0).Magnitude + 0.6), color = Color3.fromRGB(120, 132, 140):Lerp(Palette.Ancient.Stone, t0), material = if t0 < 0.4 then Enum.Material.Ice else Enum.Material.Slate, cframe = CFrame.lookAt(mid, p1) * CFrame.new(0, -4.2, 0) })
		floorSeg.Parent = parent

		if i % 2 == 0 then
			-- Metal support beams the expedition installed.
			for _, side in { -1, 1 } do
				local beam = Kit.part({ name = "SupportBeam", size = Vector3.new(0.6, 8, 0.6), color = Palette.City.Equipment, material = Enum.Material.Metal, cframe = CFrame.new(mid + Vector3.new(side * 4, 0, 0)) })
				beam.Parent = parent
			end
			local workLight = Kit.part({ name = "WorkLight", size = Vector3.new(0.8, 0.4, 0.8), color = Color3.fromRGB(255, 230, 190), material = Enum.Material.Neon, cframe = CFrame.new(mid + Vector3.new(0, 3.6, 0)) })
			workLight.Parent = parent
			Kit.light(workLight, Color3.fromRGB(255, 224, 180), 16, 1.6)
		end

		if t0 > 0.35 then
			-- Ice giving way to ancient geometry: pillars flank the deeper half.
			for _, side in { -1, 1 } do
				local pillar = Kit.part({ name = "HallPillar", size = Vector3.new(1.4, 8.4, 1.4), color = Palette.Ancient.Stone, material = Enum.Material.Slate, cframe = CFrame.new(mid + Vector3.new(side * 4.2, 0, 0)), shadow = true })
				pillar.Parent = parent
			end
		end
	end

	-- Ice dripping/breaking from vibration - a few small icicle shapes.
	for i = 1, 6 do
		local pos = from:Lerp(to, rng:NextNumber(0.05, 0.35)) + Vector3.new(rng:NextNumber(-3, 3), 3.6, rng:NextNumber(-2, 2))
		local icicle = Kit.part({ name = "Icicle", size = Vector3.new(0.3, rng:NextNumber(1, 2.4), 0.3), color = Color3.fromRGB(210, 226, 232), material = Enum.Material.Ice, transparency = 0.2, cframe = CFrame.new(pos) })
		icicle.Parent = parent
	end
end

local function buildDoor(parent: Instance): { left: BasePart, right: BasePart, symbol: BasePart }
	local at = Env.Zones.Door
	local frame = Kit.part({ name = "DoorFrame", size = Vector3.new(11, 9, 1.4), color = Palette.Ancient.StonePale, material = Enum.Material.Slate, cframe = CFrame.new(at), shadow = true })
	frame.Parent = parent

	local left = Kit.part({ name = "DoorLeft", size = Vector3.new(4.7, 8.2, 0.6), color = Color3.fromRGB(224, 220, 206), material = Enum.Material.Marble, cframe = CFrame.new(at + Vector3.new(-2.4, 0, 0.5)), shadow = true })
	left.Parent = parent
	local right = Kit.part({ name = "DoorRight", size = Vector3.new(4.7, 8.2, 0.6), color = Color3.fromRGB(224, 220, 206), material = Enum.Material.Marble, cframe = CFrame.new(at + Vector3.new(2.4, 0, 0.5)), shadow = true })
	right.Parent = parent

	-- Geometric symbol panel (no readable text - abstract carved shapes).
	local symbol = Kit.part({ name = "DoorSymbol", size = Vector3.new(1.4, 1.4, 0.15), color = Palette.Ancient.Bronze, material = Enum.Material.Metal, transparency = 0.1, cframe = left.CFrame * CFrame.new(1.8, 0, -0.4) })
	symbol.Parent = parent
	for i = 1, 4 do
		local angle = (i / 4) * math.pi * 2
		local mark = Kit.part({ name = `SymbolMark{i}`, size = Vector3.new(0.35, 0.06, 0.06), color = Palette.Ancient.Teal, material = Enum.Material.Neon, transparency = 1, cframe = symbol.CFrame * CFrame.new(math.cos(angle) * 0.4, math.sin(angle) * 0.4, -0.1) })
		mark.Parent = parent
	end

	-- Battle carvings + embedded broken weapons on the flanking frame.
	for i = 1, 3 do
		local carving = Kit.part({ name = `Carving{i}`, size = Vector3.new(1.6, 2.4, 0.12), color = Palette.Ancient.StoneDark, material = Enum.Material.Slate, cframe = frame.CFrame * CFrame.new(-4.3, 1.5 - i * 0.2, 0.75) })
		carving.Parent = parent
	end
	for i = 1, 2 do
		local blade = Kit.wedge({ name = `EmbeddedWeapon{i}`, size = Vector3.new(0.2, 2.2, 0.4), color = Palette.Ancient.BronzeDark, material = Enum.Material.Metal, cframe = frame.CFrame * CFrame.new(4.3, -1 + i * 1.6, 0.7) * CFrame.Angles(0, 0, math.rad(20)) })
		blade.Parent = parent
	end

	return { left = left, right = right, symbol = symbol }
end

local function buildHall(parent: Instance, rng: Random): { BasePart }
	local from = Env.Zones.Door
	local to = Env.Zones.ChamberCenter
	local pillars = {}
	for i = 1, 5 do
		local t = i / 6
		local pos = from:Lerp(to, t)
		for _, side in { -1, 1 } do
			local pillar = Kit.part({ name = "GreatPillar", size = Vector3.new(2.2, 16, 2.2), color = Palette.Ancient.Stone, material = Enum.Material.Slate, cframe = CFrame.new(pos + Vector3.new(side * 7, 0, 0)), shadow = true })
			pillar.Parent = parent
			table.insert(pillars, pillar)
		end
		local floor = Kit.part({ name = "HallFloor", size = Vector3.new(18, 0.4, (to - from).Magnitude / 6 + 1), color = Palette.Ancient.Stone, material = Enum.Material.Slate, cframe = CFrame.lookAt(pos, to) * CFrame.new(0, -8, 0) })
		floor.Parent = parent
	end

	-- Black organic corruption frozen between mechanical structures - a few
	-- bumpy dark growths along the lower walls, foreshadowing what's below.
	for i = 1, 6 do
		local pos = from:Lerp(to, rng:NextNumber(0.2, 0.8)) + Vector3.new(rng:NextNumber(-8, 8), rng:NextNumber(-6, -2), 0)
		local growth = Kit.ball({ name = "Corruption", size = Vector3.new(rng:NextNumber(1, 2.2), rng:NextNumber(1, 2.2), rng:NextNumber(1, 2.2)), color = Color3.fromRGB(18, 14, 22), material = Enum.Material.Slate, cframe = CFrame.new(pos) })
		growth.Parent = parent
	end

	return pillars
end

local function buildChamber(parent: Instance, rng: Random): (BasePart, BasePart, { BasePart }, BasePart)
	local center = Env.Zones.ChamberCenter
	local floorY = Env.Zones.ChamberFloor.Y

	local floor = Kit.part({ name = "ChamberFloor", size = Vector3.new(70, 1, 70), color = Palette.Ancient.Stone, material = Enum.Material.Slate, cframe = CFrame.new(center.X, floorY, center.Z) })
	floor.Parent = parent
	local ceiling = Kit.part({ name = "ChamberCeiling", size = Vector3.new(70, 1, 70), color = Palette.Ancient.StoneDark, material = Enum.Material.Slate, transparency = 0.1, cframe = CFrame.new(center.X, floorY + 42, center.Z) })
	ceiling.CastShadow = false
	ceiling.Parent = parent

	-- Ring of pillars + hanging cables to sell scale around the reveal.
	for i = 1, 8 do
		local angle = (i / 8) * math.pi * 2
		local pos = center + Vector3.new(math.cos(angle) * 28, 0, math.sin(angle) * 28)
		local pillar = Kit.part({ name = "ChamberPillar", size = Vector3.new(3, 34, 3), color = Palette.Ancient.Stone, material = Enum.Material.Slate, cframe = CFrame.new(pos.X, floorY + 17, pos.Z), shadow = true })
		pillar.Parent = parent
		if i % 2 == 0 then
			local cable = Kit.part({ name = "HangingCable", size = Vector3.new(0.2, 14, 0.2), color = Color3.fromRGB(20, 20, 22), material = Enum.Material.Metal, cframe = CFrame.new(pos.X + rng:NextNumber(-2, 2), floorY + 30, pos.Z + rng:NextNumber(-2, 2)) })
			cable.Parent = parent
		end
	end

	-- Foreground machinery/platforms between the camera and Aegis Zero.
	for i = 1, 4 do
		local pos = center + Vector3.new(rng:NextNumber(-14, 14), 0, rng:NextNumber(8, 20))
		local platform = Kit.part({ name = "Platform", size = Vector3.new(rng:NextNumber(4, 7), 1.4, rng:NextNumber(4, 7)), color = Palette.City.Equipment, material = Enum.Material.Metal, cframe = CFrame.new(pos.X, floorY + 0.7, pos.Z) })
		platform.Parent = parent
	end

	-- The seal ring: a broad flat emblem in the floor directly beneath where
	-- Aegis Zero kneels, with a crack pattern that lights up as it opens.
	local sealRing = Kit.cylinder({ name = "SealRing", size = Vector3.new(0.3, 16, 16), color = Palette.Ancient.Bronze, material = Enum.Material.Metal, transparency = 0.3, cframe = CFrame.new(center.X, floorY + 0.2, center.Z) }, "y")
	sealRing.Parent = parent
	local sealCrack = Kit.cylinder({ name = "SealCrack", size = Vector3.new(0.15, 13, 13), color = Palette.Aegis.Core, material = Enum.Material.Neon, transparency = 1, cframe = CFrame.new(center.X, floorY + 0.35, center.Z) }, "y")
	sealCrack.Parent = parent

	-- Foreground engineer consoles (activation-order equipment).
	for i = 1, 3 do
		local pos = center + Vector3.new(-10 + i * 8, 0, 16)
		local console = Kit.part({ name = `ActivationConsole{i}`, size = Vector3.new(2.2, 2.4, 1.4), color = Palette.City.Equipment, material = Enum.Material.Metal, cframe = CFrame.new(pos.X, floorY + 1.2, pos.Z) })
		console.Parent = parent
		local glow = Kit.part({ name = "ConsoleGlow", size = Vector3.new(1.4, 1, 0.1), color = Palette.Energy.Amber, material = Enum.Material.Neon, transparency = 0.2, cframe = console.CFrame * CFrame.new(0, 0.6, -0.75) })
		glow.Parent = parent
	end

	-- The myth wall: pictorial carvings telling the whole story before the
	-- expedition understands it - the Sovereign descending, Aegis fighting
	-- it, the alien forced under the ground, chains to the seal, both
	-- awakening together. Geometric/pictorial reliefs only, never English
	-- text, per rule 5.
	local mythWall = Kit.part({ name = "MythWall", size = Vector3.new(20, 14, 1), color = Palette.Ancient.StonePale, material = Enum.Material.Slate, cframe = CFrame.new(center + Vector3.new(-24, 0, 6), center) * CFrame.new(0, floorY + 8 - center.Y, 0) })
	mythWall.Parent = parent
	local reliefColors = { Palette.Alien.Plate, Palette.Aegis.PlateOuter, Color3.fromRGB(20, 20, 22), Palette.Ancient.Bronze, Palette.Alien.Energy }
	for i = 1, 5 do
		local relief = Kit.part({ name = `Relief{i}`, size = Vector3.new(2.6, 2.6 + rng:NextNumber(0, 2), 0.2), color = reliefColors[i], material = Enum.Material.Slate, transparency = if i == 5 then 0.3 else 0, cframe = mythWall.CFrame * CFrame.new(-8 + (i - 1) * 4, 0, -0.65) })
		relief.Parent = parent
	end

	return floor, sealRing, { sealCrack }, mythWall
end

local function buildShaftAndPrison(parent: Instance, rng: Random): (BasePart, { CFrame })
	local top = Env.Zones.ShaftTop
	local bottom = Env.Zones.PrisonCenter

	-- Shaft: a chain-covered grate in the chamber floor descending to the
	-- prison, camera-traversable without real physics.
	local grate = Kit.part({ name = "ShaftGrate", size = Vector3.new(10, 0.6, 10), color = Color3.fromRGB(30, 30, 32), material = Enum.Material.DiamondPlate, transparency = 0.1, cframe = CFrame.new(top) })
	grate.Parent = parent
	for i = 1, 4 do
		local chain = Kit.cylinder({ name = `ShaftChain{i}`, size = Vector3.new(0.35, 22, 0.35), color = Palette.Ancient.StoneDark, material = Enum.Material.Metal, cframe = CFrame.new(top + Vector3.new(math.cos(i) * 3.6, -11, math.sin(i) * 3.6)) }, "y")
		chain.Parent = parent
	end
	local shaftWall = Kit.cylinder({ name = "ShaftWall", size = Vector3.new(40, 14, 14), color = Palette.Ancient.StoneDark, material = Enum.Material.Slate, transparency = 0.35, cframe = CFrame.new(top:Lerp(bottom, 0.5)) }, "y")
	shaftWall.CastShadow = false
	shaftWall.Parent = parent

	-- Prison cavern: a huge irregular rock shell around the Sovereign, black
	-- root-corruption embedded in the walls, and a set of candidate spots
	-- (varied distance/rotation/scale) for Cast.buildWarden's frozen army -
	-- deliberately irregular, never a grid, per "vary poses/sizes/distance."
	for i = 1, 20 do
		local angle = rng:NextNumber() * math.pi * 2
		local radius = rng:NextNumber(30, 55)
		local pos = bottom + Vector3.new(math.cos(angle) * radius, rng:NextNumber(-20, 24), math.sin(angle) * radius)
		local shell = Kit.part({ name = "CavernRock", size = Vector3.new(rng:NextNumber(14, 30), rng:NextNumber(14, 30), rng:NextNumber(14, 30)), color = Color3.fromRGB(28, 24, 32), material = Enum.Material.Rock, cframe = CFrame.new(pos) * CFrame.Angles(rng:NextNumber() * math.pi, rng:NextNumber() * math.pi, 0), shadow = true })
		shell.Parent = parent
	end
	for i = 1, 10 do
		local pos = bottom + Vector3.new(rng:NextNumber(-40, 40), rng:NextNumber(-16, 20), rng:NextNumber(-40, 40))
		local root = Kit.cylinder({ name = "Corruption", size = Vector3.new(0.4, rng:NextNumber(4, 10), 0.4), color = Color3.fromRGB(14, 10, 18), material = Enum.Material.Slate, cframe = CFrame.new(pos) * CFrame.Angles(rng:NextNumber(-0.6, 0.6), 0, rng:NextNumber(-0.6, 0.6)) }, "y")
		root.Parent = parent
	end

	local prisonSpots = {}
	for i = 1, 14 do
		local angle = rng:NextNumber() * math.pi * 2
		local radius = rng:NextNumber(14, 40)
		local pos = bottom + Vector3.new(math.cos(angle) * radius, rng:NextNumber(-10, 16), math.sin(angle) * radius)
		table.insert(prisonSpots, CFrame.new(pos) * CFrame.Angles(0, rng:NextNumber() * math.pi * 2, rng:NextNumber(-0.3, 0.3)))
	end

	return grate, prisonSpots
end

function Env.build(): Handle
	local folder = Instance.new("Folder")
	folder.Name = "NorthPoleCinematic"

	local rng = Kit.rng(Palette.Quality.Seed + 501)
	Kit.resetLightBudget(26)

	buildGround(folder, rng)
	buildMountains(folder, rng)
	buildIceberg(folder, rng)
	local commTower, radarDish, redLights, portableLights, vehicles = buildResearchBase(folder, rng)
	local signalDisplay = buildCommandInterior(folder)
	buildTunnel(folder, rng)
	local door = buildDoor(folder)
	local pillars = buildHall(folder, rng)
	local _chamberFloor, sealRing, sealExtras, mythWall = buildChamber(folder, rng)
	local shaftGrate, prisonSpots = buildShaftAndPrison(folder, rng)
	local markers = buildCameraMarkers(folder)

	folder.Parent = workspace

	return {
		folder = folder,
		markers = markers,
		door = door,
		sealRing = sealRing,
		sealCrack = sealExtras[1],
		shaftGrate = shaftGrate,
		commTower = commTower,
		radarDish = radarDish,
		redLights = redLights,
		portableLights = portableLights,
		vehicles = vehicles,
		prisonSpots = prisonSpots,
		pillars = pillars,
		signalDisplay = signalDisplay,
		mythWall = mythWall,
	}
end

function Env.marker(handle: Handle, name: string): CFrame
	local part = handle.markers:FindFirstChild(name)
	if part and part:IsA("BasePart") then
		return part.CFrame
	end
	warn(`[SKY BROKE] North Pole Env: missing camera marker "{name}"`)
	return CFrame.new(Env.Zones.BaseCenter)
end

function Env.destroy(handle: Handle?)
	if handle and handle.folder then
		handle.folder:Destroy()
	end
end

return Env
